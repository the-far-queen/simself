    """graph of past states for signal routing."""

        self.nodes = np.random.normal(0, 1, (100, dim))

        n = 100
        dist = np.full(n, np.inf)
        prev = np.full(n, -1, dtype=int)
        dist[start] = 0
        visited = np.zeros(n, dtype=bool)
        for _ in range(n):
            u = int(np.argmin(dist + visited * 1e9))
            if np.isinf(dist[u]) or u == end:

            visited[u] = True

            path.append(cur)
            cur = prev[cur]

        return path

class Circuit:

    The circuit is the flow of energy.

        self.charge -= released
        self.charge = max(0.0, self.charge)
        self.potential += released
        return released

        low voltage -> magnetic field -> collapse -> high voltage spark

        self.voltage = input_signal
        self.current = self.conduct()
        self.charge_up()
        if self.charge > self.capacitance * 0.8:
